export async function enhanceWithAI(recommendations: AIRecommendation[]): Promise<AIRecommendation[]> {
  if (recommendations.length === 0) return recommendations

  const aiCacheKey = `ai:${buildCacheKey(recommendations)}`
  const { value: cachedMessages, cacheHit } = await getCached<string[]>(
    aiCacheKey,
    AI_CACHE_TTL_SECONDS,     // 7200 for AI response; but we need different TTL for template fallback
    async () => { /* ... compute: try AI, fallback to templates ... */ }
  )
  // ...
}
```

Note on template fallback TTL: The current code uses AI_CACHE_TTL_MS / 2 for template responses. Since getCached takes a single TTL, two approaches work:
1. Always use the longer TTL (7200s) — acceptable since templates don't change
2. Use a wrapper that calls getCached twice with different keys
Prefer option 1 (single TTL = AI_CACHE_TTL_MS / 1000) for simplicity — template responses are deterministic and can safely be cached at 2 hours.

       export async function enhanceWithAI(
         recommendations: AIRecommendation[]
       ): Promise<AIRecommendation[]> {
         if (recommendations.length === 0) return recommendations

         const aiCacheKey = `ai:${buildCacheKey(recommendations)}`

         const { value: messages } = await getCached<string[]>(
           aiCacheKey,
           AI_CACHE_TTL_SECONDS,
           async () => {
             // Check quota using atomic Redis counter
             const quotaKey = `quota:ai:${new Date().toISOString().split('T')[0]}`
             const currentCount = await incrementQuotaCounter(quotaKey)
             const withinQuota = currentCount <= AI_DAILY_QUOTA

             const client = getAnthropicClient()
             if (client && withinQuota) {
               try {
                 const prompt = buildPrompt(recommendations)
                 const response = await client.messages.create({
                   model: 'claude-haiku-4-5-20251001',
                   max_tokens: 150 * recommendations.length,
                   temperature: 0.3,
                   messages: [{ role: 'user', content: prompt }],
                 })
                 const text = response.content[0].type === 'text' ? response.content[0].text : ''
                 const lines = text
                   .split('\n')
                   .map(l => l.replace(/^\d+\.\s*/, '').trim())
                   .filter(l => l.length > 0)
                 if (lines.length >= recommendations.length) {
                   return lines
                 }
               } catch (error) {
                 // Swallow — fall through to template fallback below
               }
             }

             // Template fallback (no AI or quota exceeded or parse failure)
             return generateTemplateMessages(recommendations)
           }
         )

         return recommendations.map((rec, i) => ({
           ...rec,
           message: messages[i] ?? rec.message,
         }))
       }
       ```

    Note: incrementQuotaCounter is called INSIDE the compute function. This is intentional — getCached only calls compute on a cache miss. A quota increment only happens when the AI response is not cached, which is the correct behavior.
